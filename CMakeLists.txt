cmake_minimum_required(VERSION 3.20)
project(Timelapse VERSION 0.1.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /permissive- /utf-8 /EHsc)
add_library(lapse_core STATIC src/core.cpp src/capture.cpp src/camera_host.cpp src/encoder.cpp)
target_include_directories(lapse_core PUBLIC src)
target_link_libraries(lapse_core PUBLIC mfplat mfreadwrite mfuuid mf ole32 oleaut32 gdi32 user32 shlwapi shell32)
add_executable(Timelapse WIN32 src/main.cpp src/engine.cpp src/app.rc)
target_link_libraries(Timelapse PRIVATE lapse_core comctl32 comdlg32 shell32 dwmapi)
target_link_options(Timelapse PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/app.manifest)
set_target_properties(Timelapse PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/release")
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE lapse_core)
add_test(NAME core_tests COMMAND core_tests)
add_executable(encoder_tests tests/encoder_tests.cpp)
target_link_libraries(encoder_tests PRIVATE lapse_core)
add_test(NAME encoder_tests COMMAND encoder_tests)
add_executable(engine_tests tests/engine_tests.cpp src/engine.cpp)
target_link_libraries(engine_tests PRIVATE lapse_core)
add_test(NAME engine_tests COMMAND engine_tests)
set_tests_properties(engine_tests PROPERTIES TIMEOUT 45)
add_executable(engine_recovery_tests tests/engine_recovery_tests.cpp src/engine.cpp)
target_link_libraries(engine_recovery_tests PRIVATE lapse_core)
add_test(NAME engine_recovery_tests COMMAND engine_recovery_tests)
set_tests_properties(engine_recovery_tests PROPERTIES TIMEOUT 45)
add_executable(camera_host_tests tests/camera_host_tests.cpp)
target_link_libraries(camera_host_tests PRIVATE lapse_core)
add_test(NAME camera_host_tests COMMAND camera_host_tests)
set_tests_properties(camera_host_tests PROPERTIES TIMEOUT 35)
add_executable(camera_transport_tests tests/camera_transport_tests.cpp src/camera_host.cpp src/core.cpp)
target_include_directories(camera_transport_tests PRIVATE src)
target_link_libraries(camera_transport_tests PRIVATE mfplat mfreadwrite mfuuid mf ole32 user32 gdi32 shell32)
add_test(NAME camera_transport_tests COMMAND camera_transport_tests)
set_tests_properties(camera_transport_tests PROPERTIES TIMEOUT 25)
add_executable(engine_storage_tests tests/engine_storage_tests.cpp src/engine.cpp)
target_link_libraries(engine_storage_tests PRIVATE lapse_core)
add_test(NAME engine_storage_tests COMMAND engine_storage_tests)
set_tests_properties(engine_storage_tests PROPERTIES TIMEOUT 45)
add_executable(recording_soak_tests tests/recording_soak_tests.cpp)
target_link_libraries(recording_soak_tests PRIVATE lapse_core)
add_test(NAME recording_soak_tests COMMAND recording_soak_tests)
set_tests_properties(recording_soak_tests PROPERTIES TIMEOUT 90)
