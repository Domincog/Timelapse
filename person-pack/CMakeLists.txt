cmake_minimum_required(VERSION 3.20)
project(TimelapsePersonPack LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The optional worker requires MSVC Windows x64")
endif()
set(NCNN_SOURCE_DIR "" CACHE PATH "Pinned ncnn 20260526 full-source directory")
set(PERSON_MODEL_DIR "" CACHE PATH "Pinned NanoDet v0.4.0 ncnn-nanodet-m directory")
option(PERSON_PACK_TESTS "Build model/protocol fixtures (not distributed)" ON)
if(NOT EXISTS "${NCNN_SOURCE_DIR}/src/net.cpp")
  message(FATAL_ERROR "Set NCNN_SOURCE_DIR to the verified pinned source archive")
endif()
set(PERSON_PARAM "${PERSON_MODEL_DIR}/nanodet_m.param")
set(PERSON_WEIGHTS "${PERSON_MODEL_DIR}/nanodet_m.bin")
file(SHA256 "${PERSON_PARAM}" param_hash)
file(SHA256 "${PERSON_WEIGHTS}" weights_hash)
if(NOT param_hash STREQUAL "3a7a56919798fb3fb508aaea05cdf5c3426fa69c1394c85de17e4bdc9d7bf971" OR
   NOT weights_hash STREQUAL "04b6e85de8929f0f1603d4ad0c16d065ed8ccec56cbaf23e6d41b0a4d97b1073")
  message(FATAL_ERROR "Model bytes do not match the pinned official release")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(NCNN_BUILD_WITH_STATIC_CRT ON CACHE BOOL "" FORCE)
foreach(flag NCNN_SHARED_LIB NCNN_VULKAN NCNN_OPENMP NCNN_SIMPLEOMP NCNN_THREADS NCNN_BUILD_TOOLS NCNN_BUILD_EXAMPLES NCNN_BUILD_BENCHMARK NCNN_BUILD_TESTS NCNN_C_API NCNN_PLATFORM_API NCNN_PIXEL_ROTATE NCNN_PIXEL_AFFINE NCNN_PIXEL_DRAWING NCNN_INT8 NCNN_BF16 NCNN_XOP NCNN_AVX512 NCNN_AVX512VNNI NCNN_AVX512BF16 NCNN_AVX512FP16 NCNN_AVXVNNI NCNN_AVXVNNIINT8 NCNN_AVXVNNIINT16 NCNN_AVXNECONVERT NCNN_ENABLE_LTO)
  set(${flag} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(flag NCNN_RUNTIME_CPU NCNN_PIXEL NCNN_STDIO NCNN_STRING NCNN_SSE2 NCNN_AVX NCNN_FMA NCNN_F16C NCNN_AVX2)
  set(${flag} ON CACHE BOOL "" FORCE)
endforeach()
set(NCNN_DISABLE_RTTI OFF CACHE BOOL "" FORCE)
set(NCNN_DISABLE_EXCEPTION OFF CACHE BOOL "" FORCE)
set(needed binaryop concat convolution convolutiondepthwise crop input interp permute pooling reshape shufflechannel sigmoid slice split packing padding cast relu scale flatten innerproduct)
file(READ "${NCNN_SOURCE_DIR}/src/CMakeLists.txt" source_cmake)
string(REGEX MATCHALL "ncnn_add_layer\\([A-Za-z0-9_]+" operators "${source_cmake}")
foreach(op IN LISTS operators)
  string(REPLACE "ncnn_add_layer(" "" op "${op}")
  string(TOLOWER "${op}" op)
  if(op IN_LIST needed)
    set(WITH_LAYER_${op} ON CACHE BOOL "" FORCE)
  else()
    set(WITH_LAYER_${op} OFF CACHE BOOL "" FORCE)
  endif()
endforeach()
add_subdirectory("${NCNN_SOURCE_DIR}" ncnn)
set(PERSON_PACK_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}")
configure_file(model.rc.in model.rc @ONLY)
function(person_flags target)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE NOMINMAX UNICODE _UNICODE)
  target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/WX;/O2;/Gy;/EHsc>")
  target_link_options(${target} PRIVATE /OPT:REF /OPT:ICF)
  target_link_libraries(${target} PRIVATE ncnn)
endfunction()
add_executable(TimelapsePersonWorker worker.cpp model.cpp "${CMAKE_CURRENT_BINARY_DIR}/model.rc")
person_flags(TimelapsePersonWorker)
if(PERSON_PACK_TESTS)
  enable_testing()
  add_library(person_model_fixture SHARED tests/model_fixture.cpp model.cpp "${CMAKE_CURRENT_BINARY_DIR}/model.rc")
  person_flags(person_model_fixture)
  add_executable(person_model_tests tests/model_tests.cpp model.cpp "${CMAKE_CURRENT_BINARY_DIR}/model.rc")
  person_flags(person_model_tests)
  add_test(NAME person_model_tests COMMAND person_model_tests)
  add_executable(person_worker_tests tests/worker_tests.cpp)
  person_flags(person_worker_tests)
  add_test(NAME person_worker_tests COMMAND person_worker_tests $<TARGET_FILE:TimelapsePersonWorker>)
  set_tests_properties(person_model_tests person_worker_tests PROPERTIES TIMEOUT 20)
endif()
